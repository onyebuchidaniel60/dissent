## Build commands

```bash
# 1. narration, per segment (text from demo/NARRATION.md)
/tmp/opencode/piper-venv/bin/piper \
  -m /tmp/opencode/piper-ljspeech/en_US-ljspeech-high.onnx \
  -f S1.wav --length-scale 0.98 --sentence-silence 0.22 <<< "Segment text"

# 2. voice concat: 1.5s intro + 0.55s gaps + 1.0s outro
#    (demo/timings.json is generated by the same script)

# 3. capture, 1440x900, current local build against the live testnet DB
node demo/record-phase17.mjs
#    -> /tmp/opencode/phase17-segments/*.webm
#    -> demo/phase17-segments.json

# 4. trim each segment to the measured narration duration
ffmpeg -i segment.webm -ss <trim> -t <duration> -vf scale=1440:900,fps=30 \
  -an -c:v libx264 -preset veryfast -crf 18 -pix_fmt yuv420p segment.mp4

# 5. concat silent video
ffmpeg -f concat -safe 0 -i concat.txt -c copy /tmp/opencode/phase17-silent.mp4

# 6. music mix
ffmpeg -i voice.wav -i Floating_Cities.mp3 \
  -filter_complex "[1:a]atrim=0:144.207,volume=0.1,afade=t=in:d=1.5,afade=t=out:st=142.207:d=2[bg];[0:a]volume=1.0[v];[v][bg]amix=inputs=2:duration=longest:dropout_transition=2" \
  -t 144.207 -c:a aac -b:a 192k demo/audio-final.m4a

# 7. captions + mux (ASS captions, 1440x900 PlayRes)
ffmpeg -i /tmp/opencode/phase17-silent.mp4 -i demo/audio-final.m4a \
  -vf "ass=demo/phase17-captions.ass" \
  -c:v libx264 -preset medium -crf 20 -pix_fmt yuv420p -c:a aac -b:a 192k \
  -movflags +faststart -shortest demo/dissent-demo.mp4
```
